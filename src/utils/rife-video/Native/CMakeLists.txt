cmake_minimum_required(VERSION 3.16)
project(RifeNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_SIMPLEVK OFF CACHE BOOL "" FORCE)
set(NCNN_VULKAN_ONLINE_SPIRV ON CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_INT8 OFF CACHE BOOL "" FORCE)
add_subdirectory("${RIFE_SOURCE}/ncnn" ncnn)
file(GLOB SHADERS "${RIFE_SOURCE}/*.comp")
foreach(SHADER IN LISTS SHADERS)
  get_filename_component(NAME "${SHADER}" NAME_WE)
  set(HEADER "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.comp.hex.h")
  add_custom_command(OUTPUT "${HEADER}"
    COMMAND ${CMAKE_COMMAND} -DSHADER_SRC=${SHADER} -DSHADER_COMP_HEADER=${HEADER}
      -P "${RIFE_SOURCE}/generate_shader_comp_header.cmake" DEPENDS "${SHADER}")
  list(APPEND HEADERS "${HEADER}")
endforeach()
# Upstream discards model-load errors. Propagate them across our C error boundary.
file(READ "${RIFE_SOURCE}/rife.cpp" RIFE_CODE)
string(REPLACE "net.load_param(parampath);" "if (net.load_param(parampath)) throw std::runtime_error(\"invalid NCNN param\");" RIFE_CODE "${RIFE_CODE}")
string(REPLACE "net.load_model(modelpath);" "if (net.load_model(modelpath)) throw std::runtime_error(\"invalid NCNN weights\");" RIFE_CODE "${RIFE_CODE}")
file(READ "${RIFE_SOURCE}/rife.h" RIFE_HEADER)
string(REPLACE "load(const std::string& modeldir)" "load(const std::string& modeldir, bool fp16_math = false)" RIFE_HEADER "${RIFE_HEADER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rife.h" "${RIFE_HEADER}")
string(REPLACE "RIFE::load(const std::string& modeldir)" "RIFE::load(const std::string& modeldir, bool fp16_math)" RIFE_CODE "${RIFE_CODE}")
string(REPLACE "opt.use_fp16_arithmetic = false;" "opt.use_fp16_arithmetic = fp16_math;" RIFE_CODE "${RIFE_CODE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rife_checked.cpp" "#include <stdexcept>\n${RIFE_CODE}")
# This pinned NCNN removed Vulkan pack8; upstream RIFE still tries to compile its
# obsolete shader, whose afpvec8 macros no longer exist. NCNN now emits pack1/4.
file(READ "${RIFE_SOURCE}/warp.cpp" WARP_CODE)
string(REPLACE "if (opt.use_fp16_packed)" "if (false) // NCNN ec19da2b has no Vulkan pack8" WARP_CODE "${WARP_CODE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/warp_compatible.cpp" "${WARP_CODE}")
add_library(rife_native SHARED bridge.cpp "${CMAKE_CURRENT_BINARY_DIR}/rife_checked.cpp" "${CMAKE_CURRENT_BINARY_DIR}/warp_compatible.cpp" ${HEADERS})
target_include_directories(rife_native PRIVATE ../Sources/CNCNN/include "${CMAKE_CURRENT_BINARY_DIR}" "${RIFE_SOURCE}")
target_link_libraries(rife_native PRIVATE ncnn "-framework Metal" "-framework Foundation" "-framework QuartzCore" "-framework IOSurface" "-framework IOKit" "-framework CoreGraphics" "-framework AppKit")
set_target_properties(rife_native PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
